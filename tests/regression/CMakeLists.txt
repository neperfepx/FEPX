# the FEPX executable on a copy of the case in the build tree (work/<case>) with

# reference/fepx/<case>/, using tests/tools/compare_sim.py. Until step 9.13 most cases were compared
# with v2-fixed's results; they were converted to FEPX references generated from ac211915 (tag

# fepx_add_regression(<name> RANKS <n> MESH <file> [COMPARE_ARGS <arg>...] ...)
#   name          case name: inputs in cases/<name>, test regression_fepx_<name>, reference
#                 reference/fepx/<name> (regenerated by FEPX_REGRESSION_UPDATE)
#   RANKS         number of MPI ranks
#   MESH          mesh file copied as simulation.msh
#   COMPARE_ARGS  extra options of tests/tools/compare_sim.py (tolerances, ignored files)
#   REFERENCE <case>  the reference is that of another case (reference/fepx/<case>), never
#                 updated by this test: the same results expected from other inputs or another
#                 number of ranks
#   SAME_AS <case>  the reference is the results of regression_fepx_<case> in this build (its work
#                 directory; a fixture run first), never updated: for exact comparisons of two
#                 settings that must give the same results (step 7.3, Q35)
#   FIXTURE       this case's run is a fixture for SAME_AS cases
#   SOURCE <case> the inputs are those of cases/<case> (default: cases/<name>), e.g. to run a case
#                 on another number of ranks (step 7.4a)
#   CASE_DIR <dir>  the inputs are those of <dir> (an absolute path; instead of SOURCE), e.g. the
#                 manual's tutorials in tutorials/ (step 8.5)
#   ORI_FILE, PHASE_FILE  files copied as simulation.ori, simulation.phase (step 8.5)
#   REQUIRES <case>  a restart: this case continues the simulation.sim of regression_fepx_<case>
#                 (a fixture run first; step 8.5)
#   TUTORIAL      a tutorial case (step 8.5): label "tutorial" (outside the default runs,
#                 `ctest -LE tutorial`), time limit fepx_timeout_3600
function(fepx_add_regression name)
  cmake_parse_arguments(ARG "FIXTURE;TUTORIAL"
                        "RANKS;MESH;REFERENCE;SAME_AS;SOURCE;CASE_DIR;ORI_FILE;PHASE_FILE;REQUIRES"
                        "COMPARE_ARGS" ${ARGN})
  fepx_mesh_for_ranks(ARG_MESH ${ARG_MESH} ${ARG_RANKS})
  set(prereq_arg "")
  if(ARG_REQUIRES)
    set(prereq_arg -DPREREQ_SIM_DIR=${CMAKE_CURRENT_BINARY_DIR}/work/${ARG_REQUIRES}/simulation.sim)
  endif()
  if(NOT ARG_SOURCE)
    set(ARG_SOURCE ${name})
  endif()
  if(NOT ARG_CASE_DIR)
    set(ARG_CASE_DIR ${FEPX_CASES_DIR}/${ARG_SOURCE})
  endif()
  if(ARG_SAME_AS)
    set(reference_dir ${CMAKE_CURRENT_BINARY_DIR}/work/${ARG_SAME_AS}/simulation.sim)
    set(read_only TRUE)
  elseif(ARG_REFERENCE)
    set(reference_dir ${FEPX_REFERENCE_DIR}/${ARG_REFERENCE})
    set(read_only TRUE)
  else()
    set(reference_dir ${FEPX_REFERENCE_DIR}/${name})
    set(read_only FALSE)
  endif()
  string(REPLACE ";" "|" compare_args "${ARG_COMPARE_ARGS}")
  add_test(NAME regression_fepx_${name}
           COMMAND ${CMAKE_COMMAND}
                   -DEXECUTABLE=$<TARGET_FILE:fepx>
                   -DCASE_SOURCE_DIR=${ARG_CASE_DIR}
                   -DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}/work/${name}
                   -DMPIEXEC=${MPIEXEC_EXECUTABLE}
                   -DMPIEXEC_NUMPROC=${MPIEXEC_NUMPROC_FLAG}
                   -DNUM_RANKS=${ARG_RANKS}
                   "-DMESH_FILE=${ARG_MESH}"
                   "-DORI_FILE=${ARG_ORI_FILE}"
                   "-DPHASE_FILE=${ARG_PHASE_FILE}"
                   "-DMPIEXEC_EXTRA=${FEPX_MPIEXEC_EXTRA_FLAGS_TEXT}"
                   ${prereq_arg}
                   -DREFERENCE_DIR=${reference_dir}
                   -DREFERENCE_READ_ONLY=${read_only}
                   -DPYTHON=${Python3_EXECUTABLE}
                   -DCOMPARE_SCRIPT=${FEPX_ROOT_DIR}/tests/tools/compare_sim.py
                   "-DCOMPARE_ARGS=${compare_args}"
                   -P ${CMAKE_CURRENT_SOURCE_DIR}/run_case.cmake)
  set_tests_properties(regression_fepx_${name} PROPERTIES
                       PROCESSORS ${ARG_RANKS}
                       TIMEOUT ${fepx_timeout_300}
                       LABELS "regression;fepx")
  if(ARG_TUTORIAL)
    set_tests_properties(regression_fepx_${name} PROPERTIES
                         TIMEOUT ${fepx_timeout_3600}
                         LABELS "regression;fepx;tutorial")
  endif()
  if(ARG_FIXTURE)
    set_tests_properties(regression_fepx_${name} PROPERTIES FIXTURES_SETUP run_${name})
  endif()
  if(ARG_REQUIRES)
    set_tests_properties(regression_fepx_${ARG_REQUIRES} PROPERTIES
                         FIXTURES_SETUP run_${ARG_REQUIRES})
    set_tests_properties(regression_fepx_${name} PROPERTIES FIXTURES_REQUIRED run_${ARG_REQUIRES})
  endif()
  if(ARG_SAME_AS)
    set_tests_properties(regression_fepx_${name} PROPERTIES FIXTURES_REQUIRED run_${ARG_SAME_AS})
  endif()
endfunction()

# The manual's tutorials (doc/tutorials.rst and its pages): FEPX's own inputs
# (tutorials/<inputs>, FEPX's defaults), one case per line of the case list
# tutorials/tutorials.txt of tier "test" (step 9.11; the same list as
# tutorials/tools/make_tutorial_pages.sh, which makes the pages' figures from these inputs), so that every
# input the manual shows is run.
set(manual_tutorial_dir ${FEPX_ROOT_DIR}/tutorials)
file(STRINGS ${manual_tutorial_dir}/tutorials.txt manual_tutorial_lines REGEX "^[^#]")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${manual_tutorial_dir}/tutorials.txt)
foreach(line IN LISTS manual_tutorial_lines)
  # Split the line into its columns: case inputs mesh ori requires ranks tier.
  string(REGEX REPLACE "[ \t]+" ";" columns "${line}")
  list(LENGTH columns num_columns)
  if(NOT num_columns EQUAL 7)
    message(FATAL_ERROR "tutorials.txt: 7 columns expected: ${line}")
  endif()
  list(GET columns 0 case)
  list(GET columns 1 inputs)
  list(GET columns 2 mesh)
  list(GET columns 3 ori)
  list(GET columns 4 requires)
  list(GET columns 5 ranks)
  list(GET columns 6 tier)
  if(NOT tier STREQUAL "test")
    continue()
  endif()
  # The optional arguments: orientation file, phase file, prerequisite run.
  set(extra_args "")
  if(ori MATCHES "^inputs:(.*)$")
    list(APPEND extra_args ORI_FILE ${manual_tutorial_dir}/${inputs}/${CMAKE_MATCH_1})
  elseif(NOT ori STREQUAL "-")
    list(APPEND extra_args ORI_FILE ${mesh_dir}/${ori})
  endif()
  if(EXISTS ${manual_tutorial_dir}/${inputs}/simulation.phase)
    list(APPEND extra_args PHASE_FILE ${manual_tutorial_dir}/${inputs}/simulation.phase)
  endif()
  set(compare_args ${fepx_own_tol})
  if(NOT requires STREQUAL "-")
    list(APPEND extra_args REQUIRES manual_tutorial_${requires})
    # The steps of the run continued are compared by the prerequisite's own case.
    if(case STREQUAL "restart_cycle2")
      list(APPEND compare_args --ignore *.step0 --ignore *.step1 --ignore *.step2)
    endif()
  endif()
  fepx_add_regression(manual_tutorial_${case} RANKS ${ranks} TUTORIAL
                             CASE_DIR ${manual_tutorial_dir}/${inputs}
                             MESH ${mesh_dir}/${mesh} ${extra_args}
                             COMPARE_ARGS ${compare_args})
endforeach()
